## 부록 A: 주요 쿼리 원문

```sql
-- A1. 최근 3일 스크리닝 거부 사유
SELECT date, reject_reason, COUNT(*) AS cnt
FROM screening_log
WHERE date >= '2026-04-21' AND passed = 0
GROUP BY date, reject_reason
ORDER BY date DESC, cnt DESC;

-- A2. MIN_FINAL_SCORE 컷 시뮬레이션
SELECT date,
       SUM(CASE WHEN score >= 45.0 THEN 1 ELSE 0 END) AS cut_45,
       SUM(CASE WHEN score >= 42.0 THEN 1 ELSE 0 END) AS cut_42,
       SUM(CASE WHEN score >= 40.0 THEN 1 ELSE 0 END) AS cut_40,
       SUM(CASE WHEN score >= 38.0 THEN 1 ELSE 0 END) AS cut_38,
       COUNT(*) AS total_filter_pass
FROM screening_log
WHERE date >= date('now','-14 days','localtime') AND passed = 1
GROUP BY date;

-- A3. 테마별 통과율 (최근 14일)
SELECT theme,
       COUNT(*) AS screened,
       SUM(CASE WHEN passed = 1 THEN 1 ELSE 0 END) AS passed_cnt,
       ROUND(100.0 * SUM(CASE WHEN passed = 1 THEN 1 ELSE 0 END) / COUNT(*), 2) AS pass_rate
FROM screening_log
WHERE date >= date('now','-14 days','localtime')
GROUP BY theme
ORDER BY pass_rate;

-- A4. AI timing_score 분포 (최근 60일 매도)
SELECT json_extract(ai_review,'$.timing_score') AS ts,
       COUNT(*) AS cnt,
       ROUND(AVG(profit_rate),2) AS avg_profit,
       ROUND(AVG(max_profit_during_hold),2) AS avg_max
FROM trade_reviews
WHERE sell_date >= date('now','-60 days','localtime')
  AND ai_review IS NOT NULL
GROUP BY ts ORDER BY ts;
```
